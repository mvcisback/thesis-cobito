  1. Concepts and their Representations

\def\universe{\mathcal{U}}
\def\concept{\varphi}
\def\ConceptClass{\Phi}
\def\alphabet{\Sigma}
\def\emptystring{\epsilon}
\def\Nat{\mathbb{N}}
\def\concat{\cdot}
